--user_language=5

# Distancia de dibujado dinamica: baja un poco cuando el S24 Ultra no llega a
# los FPS objetivo y la recupera cuando hay margen (evita tirones).
--gta4_dynamic_draw_distance=true
--gta4_dynamic_draw_distance_min=0.80
--gta4_shadow_distance_scale=1.0
--gta4_reflection_capture_distance=near
#
# Densidad de trafico y peatones: 0.5 (original 0.4). El S24 Ultra tiene
# 12 GB de RAM y CPU de sobra; mas alto aumenta el calor en sesiones largas.
--gta4_traffic_density_scale=0.5
--gta4_parked_car_density_scale=0.5
--gta4_ped_density_scale=0.5
--gta4_scenario_ped_density_scale=0.5
--gta4_profile_native_detailed_cpu=false
--gta4_profile_native_detailed_gpu=false
--gta4_native_max_queued_frames=3
# The 30 FPS guard (environment reflection rate only now) is off by default.
--gta4_fps_guard=0
#
# Estabilidad / temperatura:
# - Indica a Android cuanto trabajo tiene cada fotograma para que la CPU no
#   vaya siempre al maximo (menos calor y bateria, menos estrangulamiento).
--android_perf_hint=true
# - Mientras espera el siguiente fotograma, la CPU duerme en lugar de girar.
--gta4_frame_wait_sleep=true
# - Protector termico: temperaturas de GPU (C) a las que baja y restaura la
#   distancia de dibujado si el sistema no da datos de margen termico.
--android_thermal_guard_hot_c=72
--android_thermal_guard_cool_c=62
#
# Calidad de texturas: filtrado anisotropico 4x (casi gratis en la Adreno 750,
# texturas del suelo mucho mas nitidas). Borra la linea si prefieres decidirlo
# desde el menu del juego.
--gta4_anisotropic_filtering=4x
